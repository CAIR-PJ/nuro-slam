cmake_minimum_required(VERSION 3.20)
project(cmax_slam_ros2 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui calib)
find_package(Eigen3 REQUIRED CONFIG)
find_package(GSL REQUIRED)
find_package(glog REQUIRED CONFIG)
# Upstream GSL code has no OpenMP pragmas: do not pass unsupported -fopenmp to Apple Clang.
message(STATUS "CMax build: OpenCV ${OpenCV_VERSION}; Eigen ${Eigen3_VERSION}; ${CMAKE_SYSTEM_PROCESSOR}")
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Event.msg" "msg/EventArray.msg"
  DEPENDENCIES builtin_interfaces std_msgs)
file(GLOB SOURCES CONFIGURE_DEPENDS src/frontend/*.cpp src/backend/*.cpp src/utils/*.cpp src/*.cpp)
add_executable(cmax_slam_node ${SOURCES})
target_include_directories(cmax_slam_node PRIVATE include ${OpenCV_INCLUDE_DIRS}
  thirdparty/basalt-headers/include
  thirdparty/basalt-headers/thirdparty/Sophus
  thirdparty/basalt-headers/thirdparty/cereal/include)
ament_target_dependencies(cmax_slam_node rclcpp sensor_msgs geometry_msgs nav_msgs tf2_ros std_msgs)
target_link_libraries(cmax_slam_node ${OpenCV_LIBRARIES} Eigen3::Eigen GSL::gsl GSL::gslcblas glog::glog)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(cmax_slam_node "${typesupport}")
install(TARGETS cmax_slam_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
