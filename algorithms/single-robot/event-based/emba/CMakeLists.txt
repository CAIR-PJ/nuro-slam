cmake_minimum_required(VERSION 3.20)
project(emba_ros2 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(EMBA_BUILD_ROS2 "Build the native ROS2 wrapper with ament" ON)
option(EMBA_IO_TESTS_ONLY "Build only standard-library adapter tests" OFF)
include(CTest)
add_executable(emba_io_tests tests/io_tests.cpp)
target_include_directories(emba_io_tests PRIVATE include)
add_test(NAME emba_io_adapters COMMAND emba_io_tests)
if(EMBA_IO_TESTS_ONLY)
  return()
endif()
if(CMAKE_CXX_FLAGS MATCHES "fopenmp")
  message(FATAL_ERROR "OpenMP flags are disabled for this serial baseline")
endif()
find_package(Eigen3 REQUIRED CONFIG)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib)
find_package(Boost REQUIRED)
find_package(glog REQUIRED CONFIG)
find_path(FFTW_INCLUDE_DIR fftw3.h REQUIRED)
find_library(FFTW_LIBRARY NAMES fftw3 REQUIRED)
# No OpenMP, BLAS, FFTW threads or MPI runtime is linked directly.
add_library(emba_core STATIC
  src/emba/emba.cpp src/emba/model.cpp src/emba/solver.cpp
  src/utils/pose_manager.cpp src/utils/trajectory.cpp src/utils/event_pano_warper.cpp
  src/utils/eigen_utils.cpp src/utils/image_utils.cpp
  src/image_rec/laplace.cpp src/image_rec/poisson_reconstruction.cpp)
target_include_directories(emba_core PUBLIC include ${OpenCV_INCLUDE_DIRS} ${FFTW_INCLUDE_DIR} ${Boost_INCLUDE_DIRS}
  thirdparty/basalt-headers/include
  thirdparty/basalt-headers/thirdparty/Sophus
  thirdparty/basalt-headers/thirdparty/cereal/include)
target_compile_definitions(emba_core PUBLIC EIGEN_DONT_PARALLELIZE)
target_link_libraries(emba_core PUBLIC Eigen3::Eigen ${OpenCV_LIBRARIES} ${FFTW_LIBRARY} glog::glog)
add_executable(emba_cli src/native_cli.cpp)
target_link_libraries(emba_cli PRIVATE emba_core)
if(BUILD_TESTING)
  add_executable(emba_numeric_tests tests/numeric_tests.cpp)
  target_link_libraries(emba_numeric_tests PRIVATE emba_core)
  add_test(NAME emba_numerical_invariants COMMAND emba_numeric_tests)
endif()
if(EMBA_BUILD_ROS2)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  add_executable(emba_node src/ros2_node.cpp)
  target_link_libraries(emba_node PRIVATE emba_core)
  ament_target_dependencies(emba_node PUBLIC rclcpp nav_msgs geometry_msgs sensor_msgs std_msgs)
  install(TARGETS emba_cli emba_node DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
  ament_package()
else()
  install(TARGETS emba_cli DESTINATION bin)
endif()
message(STATUS "EMBA serial baseline: OpenCV ${OpenCV_VERSION}, Eigen ${Eigen3_VERSION}, FFTW ${FFTW_LIBRARY}; ROS2=${EMBA_BUILD_ROS2}")
